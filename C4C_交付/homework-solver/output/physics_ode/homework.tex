\problem{1}

\begin{stepbox}
A net force of 12 N acts on a 3.0 kg mass that is initially at rest.
Find the acceleration of the mass and the net force needed to keep it moving
at constant velocity.
\end{stepbox}

\solution

\begin{stepbox}
识别领域：mechanics
\end{stepbox}

\begin{stepbox}
$F = ma\ \Rightarrow\ a = \frac{F}{m}$
\end{stepbox}

\begin{stepbox}
$a = \frac{12.0}{3.0} = 4.0\ \mathrm{m/s^2}$
\end{stepbox}

\begin{stepbox}
量纲核对：$\mathrm{N/kg} = \mathrm{m/s^2}$ ✓
\end{stepbox}

\begin{stepbox}
物体做匀速直线运动时 $a = 0$，由 $F_{net}=ma$ 得
\end{stepbox}

\begin{stepbox}
$F_{net} = m\cdot 0 = 0\ \mathrm{N}$ —— 合力为零
\end{stepbox}

\begin{center}
a = 4.0\textbackslash{} \textbackslash{}mathrm\{m/s\textasciicircum{}2\},\textbackslash{}qquad F\_\{net\}\textbackslash{} (\textbackslash{}text\{匀速时\}) = 0\textbackslash{} \textbackslash{}mathrm\{N\}
\end{center}

$\boxed{v = 29.41995\ \mathrm{m/s},\quad h = 44.129925\ \mathrm{m}}$
